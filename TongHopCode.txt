                    if (parameters != null)

                    }

                    {
                        DataTable dt = new DataTable();
                        adapter.Fill(dt);
                        return dt;
                    }
                }
            }
        }

        // Hàm băm mật khẩu sang chuỗi SHA-256 để kiểm tra quyền Quản lý (Admin)
        public static string ComputeSha256(string rawText)
        {
            using (SHA256 sha = SHA256.Create())
            {
                byte[] bytes = sha.ComputeHash(Encoding.UTF8.GetBytes(rawText));
                StringBuilder sb = new StringBuilder();
                foreach (byte b in bytes)
                {
                    sb.Append(b.ToString("x2"));
                }
                return sb.ToString();
            }
        }

        // Kiểm tra tài khoản Quản lý (Admin) khi thực hiện thao tác nhạy cảm: Hủy món hoặc Giảm giá
        public bool VerifyManagerAccount(string username, string password, out string managerFullName)
        {
            managerFullName = "";
            string hash = ComputeSha256(password);
            string query = @"
                SELECT FullName, Role 
                FROM Users 
                WHERE Username = @User AND (PasswordHash = @Hash OR PasswordHash = @Raw) AND IsActive = 1";
            DataTable dt = DatabaseHelper.ExecuteQuery(query, new SqlParameter[]
            {
                new SqlParameter("@User", username.Trim()),
                new SqlParameter("@Hash", hash),
                new SqlParameter("@Raw", password.Trim())
            });

            // Hỗ trợ tài khoản admin mặc định khi chạy demo
            if (dt.Rows.Count == 0)
            {
                if (username.Trim().Equals("admin", StringComparison.OrdinalIgnoreCase) && password == "admin123")
                {
                    managerFullName = "Duy Anh (Quản lý)";
                    return true;
                }
                return false;
            }

            string role = dt.Rows[0]["Role"].ToString();
            managerFullName = dt.Rows[0]["FullName"].ToString();
            return role.Equals("Admin", StringComparison.OrdinalIgnoreCase);
        }

        // Lấy danh sách các món khách đã gọi tại bàn đang chọn (chỉ lấy hóa đơn chưa thanh toán Status = 0)
        public DataTable GetBillDetailsByTable(int tableId)
        {
            string query = @"
                SELECT bd.BillDetailId, p.ProductName AS [Tên món], bd.Quantity AS [SL], bd.UnitPrice AS [Đơn giá], 
                       (bd.Quantity * bd.UnitPrice) AS [Thành tiền], bd.KitchenStatus AS [Trạng thái]
                FROM BillDetails bd
                INNER JOIN Bills b ON bd.BillId = b.BillId
                INNER JOIN Products p ON bd.ProductId = p.ProductId
                WHERE b.TableId = @TableId AND b.Status = 0";
            return DatabaseHelper.ExecuteQuery(query, new SqlParameter[] { new SqlParameter("@TableId", tableId) });
        }

        // Thêm món vào bàn với trạng thái ban đầu là 'Chờ xác nhận' (Order nháp để đọc lại cho khách trước khi gửi pha chế)
        public void AddItemToTable(int tableId, int productId, int quantity)
        {
            using (SqlConnection conn = DatabaseHelper.GetConnection())
            {
                conn.Open();
                // Sử dụng SqlTransaction để đảm bảo toàn vẹn dữ liệu khi vừa tạo Bill vừa thêm BillDetails
                using (SqlTransaction trans = conn.BeginTransaction())
                {
                    try
                    {
                        int billId = -1;
                        using (SqlCommand cmdCheckBill = new SqlCommand("SELECT TOP 1 BillId FROM Bills WHERE TableId = @TableId AND Status = 0", conn, trans))
                        {
                            cmdCheckBill.Parameters.AddWithValue("@TableId", tableId);
                            object res = cmdCheckBill.ExecuteScalar();
                            if (res != null && res != DBNull.Value)
                            {
                                billId = Convert.ToInt32(res);
                            }
                        }

                        // Nếu bàn chưa có hóa đơn thì tạo hóa đơn mới và đổi trạng thái bàn sang 'Có khách'
                        if (billId == -1)
                        {
                            using (SqlCommand cmdCreateBill = new SqlCommand("INSERT INTO Bills (TableId, CheckInTime, Status) OUTPUT INSERTED.BillId VALUES (@TableId, GETDATE(), 0)", conn, trans))
                            {
                                cmdCreateBill.Parameters.AddWithValue("@TableId", tableId);
                                billId = Convert.ToInt32(cmdCreateBill.ExecuteScalar());
                            }

                            using (SqlCommand cmdUpdateTable = new SqlCommand("UPDATE Tables SET Status = N'Có khách' WHERE TableId = @TableId", conn, trans))
                            {
                                cmdUpdateTable.Parameters.AddWithValue("@TableId", tableId);
                                cmdUpdateTable.ExecuteNonQuery();
                            }
                        }

                        decimal unitPrice = 0;
                        using (SqlCommand cmdPrice = new SqlCommand("SELECT Price FROM Products WHERE ProductId = @ProductId", conn, trans))
                        {
                            cmdPrice.Parameters.AddWithValue("@ProductId", productId);
                            unitPrice = Convert.ToDecimal(cmdPrice.ExecuteScalar());
                        }

                        // Kiểm tra xem món đó đã có ở trạng thái 'Chờ xác nhận' chưa, nếu có thì cộng dồn số lượng
                        int existingDraftDetailId = -1;
                        using (SqlCommand cmdCheckItem = new SqlCommand("SELECT TOP 1 BillDetailId FROM BillDetails WHERE BillId = @BillId AND ProductId = @ProductId AND KitchenStatus = N'Chờ xác nhận'", conn, trans))
                        {
                            cmdCheckItem.Parameters.AddWithValue("@BillId", billId);
                            cmdCheckItem.Parameters.AddWithValue("@ProductId", productId);
                            object detailObj = cmdCheckItem.ExecuteScalar();
                            if (detailObj != null && detailObj != DBNull.Value)
                            {
                                existingDraftDetailId = Convert.ToInt32(detailObj);
                            }
                        }

                        if (existingDraftDetailId != -1)
                        {
                            using (SqlCommand cmdUpdateDetail = new SqlCommand("UPDATE BillDetails SET Quantity = Quantity + @Quantity WHERE BillDetailId = @BillDetailId", conn, trans))
                            {
                                cmdUpdateDetail.Parameters.AddWithValue("@Quantity", quantity);
                                cmdUpdateDetail.Parameters.AddWithValue("@BillDetailId", existingDraftDetailId);
                                cmdUpdateDetail.ExecuteNonQuery();
                            }
                        }
                        else
                        {
                            using (SqlCommand cmdInsertDetail = new SqlCommand("INSERT INTO BillDetails (BillId, ProductId, Quantity, UnitPrice, KitchenStatus) VALUES (@BillId, @ProductId, @Quantity, @UnitPrice, N'Chờ xác nhận')", conn, trans))
                            {
                                cmdInsertDetail.Parameters.AddWithValue("@BillId", billId);
                                cmdInsertDetail.Parameters.AddWithValue("@ProductId", productId);
                                cmdInsertDetail.Parameters.AddWithValue("@Quantity", quantity);
                                cmdInsertDetail.Parameters.AddWithValue("@UnitPrice", unitPrice);
                                cmdInsertDetail.ExecuteNonQuery();
                            }
                        }

                        // Cập nhật lại tổng tiền của hóa đơn
                        using (SqlCommand cmdUpdateTotal = new SqlCommand("UPDATE Bills SET TotalAmount = (SELECT ISNULL(SUM(Quantity * UnitPrice), 0) FROM BillDetails WHERE BillId = @BillId) WHERE BillId = @BillId", conn, trans))
                        {
                            cmdUpdateTotal.Parameters.AddWithValue("@BillId", billId);
                            cmdUpdateTotal.ExecuteNonQuery();
                        }

                        trans.Commit();
                    }
                    catch
                    {
                        trans.Rollback();
                        throw;
                    }
                }
            }
        }

        // Hủy 1 món khỏi hóa đơn (Lệnh DELETE này sẽ tự động kích hoạt Trigger trg_AuditBillDetailDelete trong SQL)
        public void RemoveBillDetailItem(int billDetailId, int tableId)
        {
            using (SqlConnection conn = DatabaseHelper.GetConnection())
            {
                conn.Open();
                using (SqlTransaction trans = conn.BeginTransaction())
                {
                    try
                    {
                        int billId = -1;
                        using (SqlCommand cmdGetBill = new SqlCommand("SELECT BillId FROM BillDetails WHERE BillDetailId = @DetailId", conn, trans))
                        {
                            cmdGetBill.Parameters.AddWithValue("@DetailId", billDetailId);
                            object obj = cmdGetBill.ExecuteScalar();
                            if (obj != null && obj != DBNull.Value)
                            {
                                billId = Convert.ToInt32(obj);
                            }
                        }

                        if (billId == -1) return;

                        using (SqlCommand cmdDelete = new SqlCommand("DELETE FROM BillDetails WHERE BillDetailId = @DetailId", conn, trans))
                        {
                            cmdDelete.Parameters.AddWithValue("@DetailId", billDetailId);
                            cmdDelete.ExecuteNonQuery();
                        }

                        int remainingCount = 0;
                        using (SqlCommand cmdCount = new SqlCommand("SELECT COUNT(*) FROM BillDetails WHERE BillId = @BillId", conn, trans))
                        {
                            cmdCount.Parameters.AddWithValue("@BillId", billId);
                            remainingCount = Convert.ToInt32(cmdCount.ExecuteScalar());
                        }

                        // Nếu hủy hết sạch các món trong bàn thì xóa hóa đơn rỗng và trả bàn về trạng thái 'Trống'
                        if (remainingCount == 0)
                        {
                            using (SqlCommand cmdDelBill = new SqlCommand("DELETE FROM Bills WHERE BillId = @BillId", conn, trans))
                            {
                                cmdDelBill.Parameters.AddWithValue("@BillId", billId);
                                cmdDelBill.ExecuteNonQuery();
                            }

                            using (SqlCommand cmdFreeTable = new SqlCommand("UPDATE Tables SET Status = N'Trống' WHERE TableId = @TableId", conn, trans))
                            {
                                cmdFreeTable.Parameters.AddWithValue("@TableId", tableId);
                                cmdFreeTable.ExecuteNonQuery();
                            }
                        }
                        else
                        {
                            using (SqlCommand cmdUpdateTotal = new SqlCommand("UPDATE Bills SET TotalAmount = (SELECT ISNULL(SUM(Quantity * UnitPrice), 0) FROM BillDetails WHERE BillId = @BillId) WHERE BillId = @BillId", conn, trans))
                            {
                                cmdUpdateTotal.Parameters.AddWithValue("@BillId", billId);
                                cmdUpdateTotal.ExecuteNonQuery();
                            }
                        }

                        trans.Commit();
                    }
                    catch
                    {
                        trans.Rollback();
                        throw;
                    }
                }
            }
        }

        // Nghiệp vụ CHUYỂN BÀN: Đổi khách từ bàn hiện tại sang một bàn đang TRỐNG
        public void MoveToEmptyTable(int sourceTableId, int targetTableId)
        {
            using (SqlConnection conn = DatabaseHelper.GetConnection())
            {
                conn.Open();
                using (SqlTransaction trans = conn.BeginTransaction())
                {
                    try
                    {
                        using (SqlCommand cmdMoveBill = new SqlCommand("UPDATE Bills SET TableId = @TargetTableId WHERE TableId = @SourceTableId AND Status = 0", conn, trans))
                        {
                            cmdMoveBill.Parameters.AddWithValue("@TargetTableId", targetTableId);
                            cmdMoveBill.Parameters.AddWithValue("@SourceTableId", sourceTableId);
                            cmdMoveBill.ExecuteNonQuery();
                        }

                        using (SqlCommand cmdFreeSource = new SqlCommand("UPDATE Tables SET Status = N'Trống' WHERE TableId = @SourceTableId", conn, trans))
                        {
                            cmdFreeSource.Parameters.AddWithValue("@SourceTableId", sourceTableId);
                            cmdFreeSource.ExecuteNonQuery();
                        }

                        using (SqlCommand cmdOccupyTarget = new SqlCommand("UPDATE Tables SET Status = N'Có khách' WHERE TableId = @TargetTableId", conn, trans))
                        {
                            cmdOccupyTarget.Parameters.AddWithValue("@TargetTableId", targetTableId);
                            cmdOccupyTarget.ExecuteNonQuery();
                        }

                        trans.Commit();
                    }
                    catch
                    {
                        trans.Rollback();
                        throw;
                    }
                }
            }
        }

        // Nghiệp vụ GỘP BÀN: Dồn toàn bộ món của bàn nguồn vào bàn đích (cả 2 bàn đều đang CÓ KHÁCH)
        public void MergeOccupiedTables(int sourceTableId, int targetTableId)
        {
            using (SqlConnection conn = DatabaseHelper.GetConnection())
            {
                conn.Open();
                using (SqlTransaction trans = conn.BeginTransaction())
                {
                    try
                    {
                        int sourceBillId = -1;
                        int targetBillId = -1;

                        using (SqlCommand cmdSource = new SqlCommand("SELECT TOP 1 BillId FROM Bills WHERE TableId = @TableId AND Status = 0", conn, trans))
                        {
                            cmdSource.Parameters.AddWithValue("@TableId", sourceTableId);
                            object res = cmdSource.ExecuteScalar();
                            if (res != null && res != DBNull.Value) sourceBillId = Convert.ToInt32(res);
                        }

                        using (SqlCommand cmdTarget = new SqlCommand("SELECT TOP 1 BillId FROM Bills WHERE TableId = @TableId AND Status = 0", conn, trans))
                        {
                            cmdTarget.Parameters.AddWithValue("@TableId", targetTableId);
                            object res = cmdTarget.ExecuteScalar();
                            if (res != null && res != DBNull.Value) targetBillId = Convert.ToInt32(res);
                        }

                        if (sourceBillId == -1 || targetBillId == -1) return;

                        // Chuyển toàn bộ các dòng món sang hóa đơn của bàn đích
                        using (SqlCommand cmdMoveDetails = new SqlCommand("UPDATE BillDetails SET BillId = @TargetBillId WHERE BillId = @SourceBillId", conn, trans))
                        {
                            cmdMoveDetails.Parameters.AddWithValue("@TargetBillId", targetBillId);
                            cmdMoveDetails.Parameters.AddWithValue("@SourceBillId", sourceBillId);
                            cmdMoveDetails.ExecuteNonQuery();
                        }

                        // Tính lại tổng tiền cho bàn đích
                        using (SqlCommand cmdUpdateTargetTotal = new SqlCommand("UPDATE Bills SET TotalAmount = (SELECT ISNULL(SUM(Quantity * UnitPrice), 0) FROM BillDetails WHERE BillId = @TargetBillId) WHERE BillId = @TargetBillId", conn, trans))
                        {
                            cmdUpdateTargetTotal.Parameters.AddWithValue("@TargetBillId", targetBillId);
                            cmdUpdateTargetTotal.ExecuteNonQuery();
                        }

                        // Xóa hóa đơn cũ của bàn nguồn và giải phóng bàn nguồn về 'Trống'
                        using (SqlCommand cmdDeleteOldBill = new SqlCommand("DELETE FROM Bills WHERE BillId = @SourceBillId", conn, trans))
                        {
                            cmdDeleteOldBill.Parameters.AddWithValue("@SourceBillId", sourceBillId);
                            cmdDeleteOldBill.ExecuteNonQuery();
                        }

                        using (SqlCommand cmdFreeSource = new SqlCommand("UPDATE Tables SET Status = N'Trống' WHERE TableId = @SourceTableId", conn, trans))
                        {
                            cmdFreeSource.Parameters.AddWithValue("@SourceTableId", sourceTableId);
                            cmdFreeSource.ExecuteNonQuery();
                        }

                        trans.Commit();
                    }
                    catch
                    {
                        trans.Rollback();
                        throw;
                    }
                }
            }
        }

        // Chốt thanh toán hóa đơn, lưu phương thức thanh toán (Tiền mặt / Chuyển khoản / Shopee / Grab) và trả bàn về 'Trống'
        public bool CheckoutTable(int tableId, int discountPercent, string paymentMethod, out decimal finalAmount)
        {
            finalAmount = 0;
            using (SqlConnection conn = DatabaseHelper.GetConnection())
            {
                conn.Open();
                using (SqlTransaction trans = conn.BeginTransaction())
                {
                    try
                    {
                        int billId = -1;
                        using (SqlCommand cmdGetBill = new SqlCommand("SELECT TOP 1 BillId FROM Bills WHERE TableId = @TableId AND Status = 0", conn, trans))
                        {
                            cmdGetBill.Parameters.AddWithValue("@TableId", tableId);
                            object obj = cmdGetBill.ExecuteScalar();
                            if (obj == null || obj == DBNull.Value)
                            {
                                return false;
                            }
                            billId = Convert.ToInt32(obj);
                        }

                        decimal rawTotal = 0;
                        using (SqlCommand cmdSum = new SqlCommand("SELECT ISNULL(SUM(Quantity * UnitPrice), 0) FROM BillDetails WHERE BillId = @BillId", conn, trans))
                        {
                            cmdSum.Parameters.AddWithValue("@BillId", billId);
                            rawTotal = Convert.ToDecimal(cmdSum.ExecuteScalar());
                        }

                        finalAmount = rawTotal - (rawTotal * discountPercent / 100m);

                        // Cập nhật trạng thái hóa đơn Status = 1 (Đã thanh toán) và lưu loại tiền thanh toán
                        using (SqlCommand cmdCloseBill = new SqlCommand("UPDATE Bills SET CheckOutTime = GETDATE(), DiscountPercent = @Discount, TotalAmount = @FinalAmount, PaymentMethod = @Method, Status = 1 WHERE BillId = @BillId", conn, trans))
                        {
                            cmdCloseBill.Parameters.AddWithValue("@Discount", discountPercent);
                            cmdCloseBill.Parameters.AddWithValue("@FinalAmount", finalAmount);
                            cmdCloseBill.Parameters.AddWithValue("@Method", paymentMethod);
                            cmdCloseBill.Parameters.AddWithValue("@BillId", billId);
                            cmdCloseBill.ExecuteNonQuery();
                        }

                        // Giải phóng bàn về trạng thái 'Trống' sau khi khách thanh toán rời đi
                        using (SqlCommand cmdFreeTable = new SqlCommand("UPDATE Tables SET Status = N'Trống' WHERE TableId = @TableId", conn, trans))
                        {
                            cmdFreeTable.Parameters.AddWithValue("@TableId", tableId);
                            cmdFreeTable.ExecuteNonQuery();
                        }

                        trans.Commit();
                        return true;
                    }
                    catch
                    {
                        trans.Rollback();
                        throw;
                    }
                }
            }
        }

        // Lấy danh sách món đã được bấm 'Gửi pha chế' để hiển thị trên màn hình KDS
        public DataTable GetPendingKitchenItems()
        {
            string query = @"
                SELECT bd.BillDetailId AS [Mã], t.TableName AS [Bàn / Đơn], p.ProductName AS [Tên món cần pha], 
                       bd.Quantity AS [Số lượng], FORMAT(b.CheckInTime, 'HH:mm:ss') AS [Giờ gọi], bd.KitchenStatus AS [Trạng thái]
                FROM BillDetails bd
                INNER JOIN Bills b ON bd.BillId = b.BillId
                INNER JOIN Tables t ON b.TableId = t.TableId
                INNER JOIN Products p ON bd.ProductId = p.ProductId
                WHERE b.Status = 0 AND bd.KitchenStatus = N'Chờ pha chế'
                ORDER BY bd.BillDetailId ASC";
            return DatabaseHelper.ExecuteQuery(query);
        }

        // Tự động thêm cột Email vào bảng Users nếu trong SQL chưa có
        private void EnsureEmailColumnExists()
        {
            try
            {
                object colLength = DatabaseHelper.ExecuteScalar("SELECT COL_LENGTH('Users', 'Email')");
                if (colLength == null || colLength == DBNull.Value)
                {
                    DatabaseHelper.ExecuteNonQuery("ALTER TABLE Users ADD Email VARCHAR(100) NULL");
                    DatabaseHelper.ExecuteNonQuery("UPDATE Users SET Email = 'admin@gmail.com' WHERE Username = 'admin'");
                    DatabaseHelper.ExecuteNonQuery("UPDATE Users SET Email = 'thungan1@gmail.com' WHERE Username = 'thungan1'");
                }
            }
            catch
            {
                // Bỏ qua nếu đã tồn tại
            }
        }

        public UserDTO CheckLogin(string username, string plainPassword)
        {
            string hash = AuthService.HashPassword(plainPassword);
            string query = @"
                SELECT UserId, Username, FullName, Role, IsActive, Email 
                FROM Users 
                WHERE Username = @u AND PasswordHash = @p AND IsActive = 1";

            SqlParameter[] prms = {
                new SqlParameter("@u", username),
                new SqlParameter("@p", hash)
            };

            DataTable dt = DatabaseHelper.ExecuteQuery(query, prms);
            if (dt.Rows.Count > 0)
            {
                DataRow r = dt.Rows[0];
                return new UserDTO
                {
                    UserId = Convert.ToInt32(r["UserId"]),
                    Username = r["Username"].ToString(),
                    FullName = r["FullName"].ToString(),
                    Role = r["Role"].ToString(),
                    IsActive = Convert.ToBoolean(r["IsActive"]),
                    Email = r.Table.Columns.Contains("Email") && r["Email"] != DBNull.Value ? r["Email"].ToString() : ""
                };
            }
            return null;
        }

        public bool IsUsernameOrEmailExist(string username, string email)
        {
            string query = "SELECT COUNT(1) FROM Users WHERE Username = @u OR (Email IS NOT NULL AND Email = @e)";
            SqlParameter[] prms = {
                new SqlParameter("@u", username),
                new SqlParameter("@e", email)
            };
            int count = Convert.ToInt32(DatabaseHelper.ExecuteScalar(query, prms));
            return count > 0;
        }

        public bool CheckEmailExist(string email)
        {
            string query = "SELECT COUNT(1) FROM Users WHERE Email = @e AND IsActive = 1";
            SqlParameter[] prms = { new SqlParameter("@e", email) };
            int count = Convert.ToInt32(DatabaseHelper.ExecuteScalar(query, prms));
            return count > 0;
        }

        public bool RegisterUser(string username, string plainPassword, string email, string fullName)
        {
            string hash = AuthService.HashPassword(plainPassword);
            string query = @"
                INSERT INTO Users (Username, PasswordHash, FullName, Role, IsActive, Email) 
                VALUES (@u, @p, @fn, 'Staff', 1, @e)";

            SqlParameter[] prms = {
                new SqlParameter("@u", username),
                new SqlParameter("@p", hash),
                new SqlParameter("@fn", string.IsNullOrEmpty(fullName) ? username : fullName),
                new SqlParameter("@e", email)
            };
            return DatabaseHelper.ExecuteNonQuery(query, prms) > 0;
        }

        public bool UpdatePasswordByEmail(string email, string newPlainPassword)
        {
            string hash = AuthService.HashPassword(newPlainPassword);
            string query = "UPDATE Users SET PasswordHash = @p WHERE Email = @e";
            SqlParameter[] prms = {
                new SqlParameter("@p", hash),
                new SqlParameter("@e", email)
            };
            return DatabaseHelper.ExecuteNonQuery(query, prms) > 0;
        }
    }

        private void PicCoffeeCup3D_Paint(object sender, PaintEventArgs e)
        {
            Graphics g = e.Graphics;
            g.SmoothingMode = SmoothingMode.AntiAlias;

            int offsetY = cupPressOffset + (isCupHovered ? -4 : 0);

            for (int i = 0; i < 3; i++)
            {
                float wave = (float)Math.Sin(smokePhase + i * 1.5f) * 10f;
                int x = 95 + i * 25 + (int)wave;
                int y = 45 + offsetY - (int)(Math.Cos(smokePhase + i) * 6);

                using (Pen smokePen = new Pen(Color.FromArgb(isCupHovered ? 160 : 90, 255, 245, 235), 4f))
                {
                    smokePen.StartCap = LineCap.Round;
                    smokePen.EndCap = LineCap.Round;
                    g.DrawBezier(smokePen, x, y + 40, x - 15, y + 25, x + 15, y + 10, x, y - 5);
                }
            }

            using (SolidBrush shadowBrush = new SolidBrush(Color.FromArgb(110, 0, 0, 0)))
                g.FillEllipse(shadowBrush, 35, 210, 180, 28);

            using (LinearGradientBrush saucerBrush = new LinearGradientBrush(
                new Rectangle(40, 195 + offsetY, 170, 26),
                Color.FromArgb(245, 245, 245),
                Color.FromArgb(160, 150, 140), 90f))
            {
                g.FillEllipse(saucerBrush, 40, 195 + offsetY, 170, 26);
            }

            using (Pen handlePen = new Pen(Color.FromArgb(230, 220, 210), 12f))
                g.DrawArc(handlePen, 155, 115 + offsetY, 50, 65, -70, 140);

            Rectangle cupRect = new Rectangle(60, 95 + offsetY, 130, 110);
            Color c1 = isCupHovered ? Color.FromArgb(255, 167, 38) : Color.FromArgb(215, 204, 200);
            Color c2 = isCupHovered ? Color.FromArgb(216, 67, 21) : Color.FromArgb(121, 85, 72);

            using (LinearGradientBrush cupBrush = new LinearGradientBrush(cupRect, c1, c2, 0f))
            {
                GraphicsPath path = new GraphicsPath();
                path.AddArc(60, 145 + offsetY, 130, 60, 0, 180);
                path.AddLine(60, 175 + offsetY, 60, 100 + offsetY);
                path.AddLine(60, 100 + offsetY, 190, 100 + offsetY);
                path.AddLine(190, 100 + offsetY, 190, 175 + offsetY);
                path.CloseFigure();
                g.FillPath(cupBrush, path);
            }

            using (SolidBrush rimBrush = new SolidBrush(Color.FromArgb(245, 240, 235)))
                g.FillEllipse(rimBrush, 60, 85 + offsetY, 130, 30);

            using (SolidBrush coffeeLiquid = new SolidBrush(Color.FromArgb(62, 39, 35)))
                g.FillEllipse(coffeeLiquid, 66, 89 + offsetY, 118, 22);

            using (Font f = new Font("Segoe UI", 10, FontStyle.Bold))
            using (SolidBrush textBrush = new SolidBrush(Color.White))
            {
                string txt = isCupHovered ? "BẤM ĐỂ VÀO!" : "ĐĂNG NHẬP";
                SizeF sz = g.MeasureString(txt, f);
                g.DrawString(txt, f, textBrush, 125 - sz.Width / 2, 140 + offsetY);
            }
        }

        private void PerformLogin()
        {
            string u = txtLoginUser.Text.Trim();
            string p = txtLoginPass.Text.Trim();

            if (string.IsNullOrEmpty(u) || string.IsNullOrEmpty(p))
            {
                MessageBox.Show("Bạn phải nhập Tên đăng nhập và Mật khẩu trước khi bấm vào Ly Cafe hoặc nút Đăng nhập!", "Chưa nhập tài khoản", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                txtLoginUser.Focus();
                return;
            }

            try
            {
                var user = _userDal.CheckLogin(u, p);
                if (user != null)
                {
                    animTimer.Stop();
                    this.Hide();

                    FrmMainPOS frmPos = new FrmMainPOS();
                    ApplyRolePermissionsToPOS(frmPos, user.FullName, user.Role);
                    frmPos.Show();
                }
                else
                {
                    MessageBox.Show("Sai tên đăng nhập hoặc mật khẩu!", "Lỗi đăng nhập", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show("Lỗi kết nối CSDL: \n" + ex.Message, "Lỗi SQL", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void BtnReg_Click(object sender, EventArgs e)
        {
            string u = txtRegUser.Text.Trim();
            string fn = txtRegFullName.Text.Trim();

            string p = txtRegPass.Text.Trim();
            string cf = txtRegConfirm.Text.Trim();

                MessageBox.Show("Vui lòng điền đầy đủ các trường bắt buộc (*)!", "Cảnh báo", MessageBoxButtons.OK, MessageBoxIcon.Warning);

                    txtLoginPass.Text = p;
                    ShowPanel(pnlLogin);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show("Lỗi khi đăng ký: " + ex.Message, "Lỗi", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void BtnSendOtp_Click(object sender, EventArgs e)
        {
            string email = txtForgotEmail.Text.Trim();
            if (string.IsNullOrEmpty(email))
            {
                MessageBox.Show("Vui lòng nhập địa chỉ Gmail của bạn!", "Thông báo", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            try
            {
                if (!_userDal.CheckEmailExist(email))
                {
                    MessageBox.Show("Email này chưa được đăng ký trong hệ thống quán cafe!", "Không tìm thấy", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }

                _currentOtp = AuthService.GenerateOTP();
                _resetEmail = email;
                _otpExpiry = DateTime.Now.AddMinutes(5);

                if (AuthService.SendOtpEmail(email, _currentOtp, out string err))
                {
                    if (!string.IsNullOrEmpty(err))
                        MessageBox.Show(err, "Mã OTP Test", MessageBoxButtons.OK, MessageBoxIcon.Information);
                    else
                        MessageBox.Show("Đã gửi mã OTP gồm 6 chữ số tới Gmail của bạn! Vui lòng kiểm tra hộp thư.", "Thành công", MessageBoxButtons.OK, MessageBoxIcon.Information);

                    txtOtpCode.Enabled = true;
                    txtNewPass.Enabled = true;
                    btnConfirmReset.Enabled = true;
                }
                else
                {
                    MessageBox.Show("Không thể gửi email: " + err, "Lỗi SMTP", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show("Lỗi: " + ex.Message, "Lỗi", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void BtnConfirmReset_Click(object sender, EventArgs e)
        {
            string inputOtp = txtOtpCode.Text.Trim();
            string newPass = txtNewPass.Text.Trim();

            if (DateTime.Now > _otpExpiry)
            {
                MessageBox.Show("Mã OTP đã hết hạn (quá 5 phút)! Vui lòng bấm gửi lại.", "Hết hạn", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            if (inputOtp != _currentOtp)
            {
                MessageBox.Show("Mã OTP không chính xác! Vui lòng kiểm tra lại.", "Sai OTP", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            if (string.IsNullOrEmpty(newPass) || newPass.Length < 4)
            {
                MessageBox.Show("Mật khẩu mới phải từ 4 ký tự trở lên!", "Cảnh báo", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            if (_userDal.UpdatePasswordByEmail(_resetEmail, newPass))
            {
                MessageBox.Show("Đổi mật khẩu thành công! Mời bạn đăng nhập lại.", "Hoàn tất", MessageBoxButtons.OK, MessageBoxIcon.Information);
                _currentOtp = "";
                ShowPanel(pnlLogin);
            }
        }

        private decimal FinalTotal => _currentRawTotal - (_currentRawTotal * _approvedDiscountPercent / 100m);

            SwitchChannelTab(Channel.DineIn, reloadFromDb: true);

        private static Channel GetChannel(string tableName)
        {
            if (string.IsNullOrEmpty(tableName)) return Channel.DineIn;
            bool Starts(string prefix) => tableName.StartsWith(prefix, StringComparison.OrdinalIgnoreCase);

            if (Starts("Mang về")) return Channel.TakeAway;
            if (Starts("Shopee") || Starts("XanhSM")) return Channel.ShopeeXanh;
            if (Starts("Grab")) return Channel.Grab;
            return Channel.DineIn;
        }

        private void SwitchChannelTab(Channel channel, bool reloadFromDb = false)
        {
            _currentChannel = channel;
            HighlightTabs(_mainTabButtons, channel);

            if (cboPaymentMethod != null && cboPaymentMethod.Items.Count >= 4)
            {
                if (channel == Channel.ShopeeXanh) cboPaymentMethod.SelectedIndex = 2;
                else if (channel == Channel.Grab) cboPaymentMethod.SelectedIndex = 3;
                else cboPaymentMethod.SelectedIndex = 0;
            }

            LoadTablesToBoard(reloadFromDb);
        }

        private void LoadTablesToBoard(bool reloadFromDb = false)
        {
            try
            {
                if (reloadFromDb || _cachedTables.Count == 0)
                {
                    _cachedTables = _posDal.GetAllTables();
                }

                ClearAndDispose(flpTables);
                flpTables.SuspendLayout();

                foreach (TableDTO table in _cachedTables)
                {
                    if (GetChannel(table.TableName) != _currentChannel) continue;

                    bool isEmpty = table.Status == StatusEmpty;
                    bool isSelected = table.TableId == _selectedTableId;
                    string title = isSelected
                        ? $"✔ {table.TableName}\n({table.Status})"
                        : $"{table.TableName}\n({table.Status})";

                    Button btn = MakeButton(
                        title,
                        Point.Empty,
                        new Size(128, 96),
                        isEmpty ? Theme.Green : Theme.Orange,
                        TableButton_Click);

                    btn.Margin = new Padding(6);
                    btn.Tag = table;
                    btn.FlatAppearance.BorderSize = isSelected ? 4 : 0;
                    btn.FlatAppearance.BorderColor = Theme.Brown;

                    flpTables.Controls.Add(btn);
                }

                flpTables.ResumeLayout();
            }
            catch (Exception ex)
            {
                ShowError("Lỗi tải sơ đồ bàn: " + ex.Message);
            }
        }

        private void TableButton_Click(object sender, EventArgs e)
        {
            if ((sender as Button)?.Tag is TableDTO table)
            {
                _selectedTableId = table.TableId;
                _selectedTableName = table.TableName;

                // Khôi phục % giảm giá đã duyệt của riêng bàn này
                _approvedDiscountPercent = _tableDiscounts.TryGetValue(_selectedTableId, out int savedDiscount) ? savedDiscount : 0;
                nudDiscount.Value = _approvedDiscountPercent;

                lblSelectedTable.Text = $"ĐANG PHỤC VỤ: {_selectedTableName.ToUpper()} ({table.Status})";
                picVietQR.Image = null;

                // Chỉ vẽ lại viền chọn bàn từ bộ nhớ đệm, không gọi lại DB
                LoadTablesToBoard(reloadFromDb: false);
                LoadBillForSelectedTable();
            }
        }

        private void LoadBillForSelectedTable()
        {
            if (_selectedTableId == -1) return;

            try
            {
                DataTable dt = _posDal.GetBillDetailsByTable(_selectedTableId);
                dgvBillDetails.DataSource = dt;
                if (dgvBillDetails.Columns["BillDetailId"] != null)
                {
                    dgvBillDetails.Columns["BillDetailId"].Visible = false;
                }

                _currentRawTotal = 0;
                foreach (DataRow row in dt.Rows)
                {
                    _currentRawTotal += Convert.ToDecimal(row["Thành tiền"]);
                }

                if (_currentRawTotal <= 0)
                {
                    _approvedDiscountPercent = 0;
                    _tableDiscounts.Remove(_selectedTableId);
                    nudDiscount.Value = 0;
                }

                UpdateTotalDisplay();
            }
            catch (Exception ex)
            {
                ShowError("Lỗi tải chi tiết hóa đơn: " + ex.Message);
            }
        }

        private void UpdateTotalDisplay()
        {
            lblTotalAmount.Text = _approvedDiscountPercent > 0
                ? $"TỔNG THANH TOÁN (Giảm {_approvedDiscountPercent}%): {FinalTotal:N0} VNĐ"
                : $"TỔNG THANH TOÁN: {FinalTotal:N0} VNĐ";
        }

        {
            if (_selectedTableId == -1 || dgvBillDetails.CurrentRow == null)
            {
                Warn("Vui lòng chọn một món trong bảng hóa đơn để hủy!");
                return;
            }

            int billDetailId = Convert.ToInt32(dgvBillDetails.CurrentRow.Cells["BillDetailId"].Value);
            string itemName = dgvBillDetails.CurrentRow.Cells["Tên món"].Value.ToString();

            if (!RequestManagerApprovalDialog($"Hủy món [{itemName}] tại {_selectedTableName}", out string managerName))
            {
                return;
            }

            try
            {
                _posDal.RemoveBillDetailItem(billDetailId, _selectedTableId);
                RefreshAll();
                Info($"Đã hủy món [{itemName}] (Duyệt bởi Quản lý: {managerName}).", "Đã hủy món");
            }
            catch (Exception ex)
            {
                ShowError("Lỗi khi hủy món: " + ex.Message);
            }
        }

        private void btnApplyDiscount_Click(object sender, EventArgs e)
        {
            if (!EnsureHasBill("Bàn chưa có hóa đơn để áp dụng giảm giá!")) return;

            int requestedDiscount = Convert.ToInt32(nudDiscount.Value);
            if (requestedDiscount == 0)
            {
                _approvedDiscountPercent = 0;
                _tableDiscounts.Remove(_selectedTableId);
                UpdateTotalDisplay();
                return;
            }

            if (!RequestManagerApprovalDialog($"Giảm giá {requestedDiscount}% cho {_selectedTableName}", out string managerName))
            {
                nudDiscount.Value = _approvedDiscountPercent;
                return;
            }

            _approvedDiscountPercent = requestedDiscount;
            _tableDiscounts[_selectedTableId] = requestedDiscount;
            UpdateTotalDisplay();
            Info($"Đã duyệt giảm giá {_approvedDiscountPercent}% bởi Quản lý: {managerName}!", "Thành công");
        }

        private void btnMoveTable_Click(object sender, EventArgs e)
        {
            if (!EnsureHasBill("Vui lòng chọn bàn đang có khách cần chuyển đi!")) return;

            TableDTO targetTable = ShowVisualTableSelectorDialog(isMoveToEmptyMode: true);
            if (targetTable == null) return;

            try
            {
                int oldTableId = _selectedTableId;
                _posDal.MoveToEmptyTable(oldTableId, targetTable.TableId);

                if (_tableDiscounts.TryGetValue(oldTableId, out int discount))
                {
                    _tableDiscounts.Remove(oldTableId);
                    _tableDiscounts[targetTable.TableId] = discount;
                }

                _selectedTableId = targetTable.TableId;
                _selectedTableName = targetTable.TableName;
                RefreshAll();
                Info($"Đã chuyển sang [{_selectedTableName}] thành công!", "Chuyển bàn hoàn tất");
            }
            catch (Exception ex)
            {
                ShowError("Lỗi khi chuyển bàn: " + ex.Message);
            }
        }

        private void btnMergeTable_Click(object sender, EventArgs e)
        {
            if (!EnsureHasBill("Vui lòng chọn bàn đang có khách cần gộp!")) return;

            TableDTO targetTable = ShowVisualTableSelectorDialog(isMoveToEmptyMode: false);
            if (targetTable == null) return;

            try
            {
                int sourceTableId = _selectedTableId;
                _posDal.MergeOccupiedTables(sourceTableId, targetTable.TableId);

                // Xóa giảm giá khi gộp bàn để tránh áp dụng sai mức giảm của đơn nhỏ sang đơn lớn
                _tableDiscounts.Remove(sourceTableId);
                _tableDiscounts.Remove(targetTable.TableId);
                _approvedDiscountPercent = 0;
                nudDiscount.Value = 0;

                _selectedTableId = targetTable.TableId;
                _selectedTableName = targetTable.TableName;
                RefreshAll();
                Info($"Đã gộp chung hóa đơn vào [{_selectedTableName}] thành công!", "Gộp bàn hoàn tất");
            }
            catch (Exception ex)
            {
                ShowError("Lỗi khi gộp bàn: " + ex.Message);
            }
        }

        private void btnGenerateQR_Click(object sender, EventArgs e)
        {
            if (!EnsureHasBill("Bàn / Đơn này chưa có hóa đơn cần thanh toán!")) return;

            cboPaymentMethod.SelectedIndex = 1;
            string qrUrl = SmartPaymentService.BuildVietQrUrl(FinalTotal, _selectedTableName);
            picVietQR.LoadAsync(qrUrl);
        }

        private void btnPrintBill_Click(object sender, EventArgs e)
        {
            if (!EnsureHasBill("Chưa có hóa đơn để in!")) return;

            try
            {
                string method = cboPaymentMethod.SelectedItem?.ToString() ?? "Tiền mặt";
                DataTable dtItems = _posDal.GetBillDetailsByTable(_selectedTableId);
                string receiptContent = BuildReceiptText(dtItems, method);
                ShowReceiptPreviewDialog(receiptContent);
            }
            catch (Exception ex)
            {
                ShowError("Lỗi khi tạo phiếu in: " + ex.Message);
            }
        }

        private void btnConfirmPayment_Click(object sender, EventArgs e)
        {
            if (!EnsureHasBill("Không có hóa đơn để thanh toán!")) return;

            string paymentMethod = cboPaymentMethod.SelectedItem?.ToString() ?? "Tiền mặt";
            decimal expectedTotal = FinalTotal;

            DialogResult confirm = MessageBox.Show(
                $"Xác nhận đã nhận đủ [{expectedTotal:N0} VNĐ] cho [{_selectedTableName}]?\n- Hình thức: {paymentMethod}\n- Bàn/Đơn sẽ tự động chuyển về trạng thái Trống.",
                "Xác nhận thanh toán",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Question);

            if (confirm != DialogResult.Yes) return;

            try
            {
                if (_posDal.CheckoutTable(_selectedTableId, _approvedDiscountPercent, paymentMethod, out decimal finalPaid))
                {
                    string paidTableName = _selectedTableName;
                    _tableDiscounts.Remove(_selectedTableId);
                    picVietQR.Image = null;
                    _approvedDiscountPercent = 0;
                    nudDiscount.Value = 0;
                    lblSelectedTable.Text = $"ĐÃ HOÀN TẤT: {paidTableName.ToUpper()} ({StatusEmpty})";

                    RefreshAll();
                    SmartPaymentService.PlaySuccessSound();
                    Info($"Đã thanh toán thành công {paidTableName}!\n- Số tiền: {finalPaid:N0} VNĐ\n- Nguồn tiền: {paymentMethod}\n- Trạng thái: Bàn đã Trống.", "Hoàn tất");
                }
                else
                {
                    Warn("Không thể hoàn tất thanh toán. Vui lòng kiểm tra lại trạng thái hóa đơn!");
                }
            }
            catch (Exception ex)
            {
                ShowError("Lỗi khi thanh toán: " + ex.Message);
            }
        }
    }

        private bool RequestManagerApprovalDialog(string actionTitle, out string approvedBy)
        {
            approvedBy = "";
            using (Form authForm = new Form())
            {
                authForm.Text = "Yêu cầu Quản lý phê duyệt";
                authForm.Size = new Size(420, 250);
                authForm.StartPosition = FormStartPosition.CenterParent;
                authForm.FormBorderStyle = FormBorderStyle.FixedDialog;
                authForm.MaximizeBox = false;
                authForm.MinimizeBox = false;
                authForm.BackColor = Color.White;

                Label lblHeader = new Label
                {
                    Text = $"Thao tác: {actionTitle}\n(Chỉ tài khoản Quản lý mới được phép thực hiện)",
                    Font = Theme.Bold95,
                    ForeColor = Theme.Red,
                    Location = new Point(20, 15),
                    AutoSize = true
                };

                Label lblUser = new Label { Text = "TK Quản lý:", Location = new Point(20, 72), AutoSize = true, Font = Theme.Regular10 };
                TextBox txtUser = new TextBox { Location = new Point(125, 69), Size = new Size(250, 28), Font = Theme.Regular10 };

                Label lblPass = new Label { Text = "Mật khẩu:", Location = new Point(20, 112), AutoSize = true, Font = Theme.Regular10 };
                TextBox txtPass = new TextBox { Location = new Point(125, 109), Size = new Size(250, 28), Font = Theme.Regular10, UseSystemPasswordChar = true };

                Button btnOk = MakeButton("Duyệt thao tác", new Point(125, 155), new Size(140, 36), Theme.Green);
                btnOk.DialogResult = DialogResult.OK;

                Button btnCancel = MakeButton("Hủy bỏ", new Point(275, 155), new Size(100, 36), Color.Gray, null, Theme.Regular95);
                btnCancel.DialogResult = DialogResult.Cancel;

                authForm.Controls.AddRange(new Control[] { lblHeader, lblUser, txtUser, lblPass, txtPass, btnOk, btnCancel });
                authForm.AcceptButton = btnOk;
                authForm.CancelButton = btnCancel;

                if (authForm.ShowDialog(this) != DialogResult.OK) return false;

                try
                {
                    if (_posDal.VerifyManagerAccount(txtUser.Text.Trim(), txtPass.Text, out approvedBy))
                    {
                        return true;
                    }
                }
                catch (Exception ex)
                {
                    ShowError("Lỗi xác thực tài khoản: " + ex.Message);

        private string BuildReceiptText(DataTable dtItems, string method)
        {
            StringBuilder sb = new StringBuilder();
            sb.AppendLine("        HIGHLAND COFFEE POS - NHÓM 8");
            sb.AppendLine("      ĐC: Công Viên Thủy Tinh, Thủ Đức");
            sb.AppendLine("==========================================");
            sb.AppendLine("         PHIẾU TẠM TÍNH / HÓA ĐƠN");
            sb.AppendLine($"Bàn / Đơn : {_selectedTableName}");
            sb.AppendLine($"Thời gian : {DateTime.Now:dd/MM/yyyy HH:mm:ss}");
            sb.AppendLine($"Hình thức : {method}");
            sb.AppendLine("------------------------------------------");
            sb.AppendLine(string.Format("{0,-18} {1,3} {2,9} {3,9}", "Tên món", "SL", "Đ.Giá", "T.Tiền"));
            sb.AppendLine("------------------------------------------");

            foreach (DataRow row in dtItems.Rows)
            {
                string name = row["Tên món"].ToString();
                if (name.Length > 17) name = name.Substring(0, 17);
                int qty = Convert.ToInt32(row["SL"]);
                decimal price = Convert.ToDecimal(row["Đơn giá"]);
                decimal sub = Convert.ToDecimal(row["Thành tiền"]);
                sb.AppendLine(string.Format("{0,-18} {1,3} {2,9:N0} {3,9:N0}", name, qty, price, sub));
            }

            sb.AppendLine("------------------------------------------");
            sb.AppendLine($"Tạm tính          : {_currentRawTotal:N0} VNĐ");
            sb.AppendLine($"Giảm giá          : {_approvedDiscountPercent}%");
            sb.AppendLine($"TỔNG THANH TOÁN   : {FinalTotal:N0} VNĐ");
            sb.AppendLine("==========================================");
            sb.AppendLine("    Quét mã VietQR bên dưới để thanh toán:");
            return sb.ToString();
        }
